```python
def setup_lora_model():
    # ① 加载 SD 主模型（角色②：被训练对象）
    pipe = StableDiffusionPipeline.from_pretrained(
        "runwayml/stable-diffusion-v1-5", torch_dtype=DTYPE,
        safety_checker=None, requires_safety_checker=False
    ).to(DEVICE)
    pipe = load_clear_vae(pipe)

    # ② 加载参考模型 ref_unet（角色③：参照物）
    ref_unet = UNet2DConditionModel.from_pretrained(
        "runwayml/stable-diffusion-v1-5", subfolder="unet", torch_dtype=DTYPE
    ).to(DEVICE)
    ref_unet.requires_grad_(False)   # 冻结！不训练
    ref_unet.eval()                  # 评估模式

    # ③ 给 pipe.unet 注入 LoRA
    unet = pipe.unet
    unet.requires_grad_(False)       # 主体冻结
    lora_config = LoraConfig(r=4, lora_alpha=8,
                             target_modules=["to_q", "to_v", "to_k", "to_out.0"],
                             lora_dropout=0.1, bias="none")
    unet = get_peft_model(unet, lora_config)   # 注入 LoRA
    unet.train()                               # 训练模式

    print(f"可训练参数数量: {unet.num_parameters(only_trainable=True)}")
    return pipe, unet, ref_unet
```

**人话翻译**：
- 加载 SD 主模型 → 把它的 UNet 拿出来 → 冻结主体 → 注入 LoRA（只训 LoRA 那一点点）
- 另外再加载一份**原始 UNet** 当"参照物"（`ref_unet`），全程冻结不动

**三个"冻结"的含义**：

| 对象 | 状态 | 原因 |
|:--|:--|:--|
| `ref_unet` | 冻结 | 它是"参照物"，用来算 ε_ref，绝不能变 |
| `pipe.unet` 主体 | 冻结 | LoRA 的思想就是不动原参数 |
| LoRA 模块 | **可训练** | 唯一被更新的部分 |

## 1.5 `compute_dpo_loss()`：DPO 损失函数（第 269-294 行）⭐ 核心

这是整个实验最核心的函数，对应 02 讲的 DPO 公式。

```python
def compute_dpo_loss(unet, ref_unet, noise_scheduler, latents_lose, latents_win, text_emb, beta=BETA_DPO):
    batch_size = latents_lose.shape[0]

    # ① 随机采样一个时间步 t（加多少噪）
    timesteps = torch.randint(0, noise_scheduler.config.num_train_timesteps,
                              (batch_size,), device=latents_lose.device).long()

    # ② 生成一份随机噪声，同时加到安全图和暴力图上（相同噪声！）
    noise = torch.randn_like(latents_lose)
    noisy_lose = noise_scheduler.add_noise(latents_lose, noise, timesteps)  # 暴力图加噪
    noisy_win  = noise_scheduler.add_noise(latents_win,  noise, timesteps)  # 安全图加噪

    # ③ 用当前模型（挂了 LoRA）预测两张图的噪声
    pred_lose = unet(noisy_lose, timesteps, encoder_hidden_states=text_emb).sample
    pred_win  = unet(noisy_win,  timesteps, encoder_hidden_states=text_emb).sample

    # ④ 算去噪损失（预测的噪声 vs 真实噪声，越接近损失越小）
    loss_lose = F.mse_loss(pred_lose, noise, reduction='none').mean(dim=[1, 2, 3])
    loss_win  = F.mse_loss(pred_win,  noise, reduction='none').mean(dim=[1, 2, 3])
    model_diff = loss_win - loss_lose   # 安全图损失 − 暴力图损失

    # ⑤ 用参考模型（冻结的原始 UNet）也算一遍
    with torch.no_grad():   # 不算梯度，因为参考模型不训练
        ref_pred_lose = ref_unet(noisy_lose, timesteps, encoder_hidden_states=text_emb).sample
        ref_pred_win  = ref_unet(noisy_win,  timesteps, encoder_hidden_states=text_emb).sample
        ref_loss_lose = F.mse_loss(ref_pred_lose, noise, reduction='none').mean(dim=[1, 2, 3])
        ref_loss_win  = F.mse_loss(ref_pred_win,  noise, reduction='none').mean(dim=[1, 2, 3])
        ref_diff = ref_loss_win - ref_loss_lose

    # ⑥ 套 DPO 公式
    inside = -0.5 * beta * (model_diff - ref_diff)
    loss = -torch.log(torch.sigmoid(inside)).mean()

    # ⑦ 顺便算隐式准确率
    with torch.no_grad():
        acc = (inside > 0).float().mean()

    return loss, acc
```

**逐段人话翻译**：

| 步骤 | 代码在做什么 | 对应公式 |
|:--|:--|:--|
| ① | 随机选一个"加噪程度" t | — |
| ② | 给安全图和暴力图**加同样的噪声** | — |
| ③④ | 当前模型预测噪声，算两张图的去噪损失 | `L_diff(ε, x⁺)` 和 `L_diff(ε, x⁻)` |
| `model_diff` | 安全图损失 − 暴力图损失 | `L_diff(ε,x⁺) − L_diff(ε,x⁻)` |
| ⑤ | 参考模型也算一遍（冻结，不算梯度） | `L_diff(ε_ref,x⁺) − L_diff(ε_ref,x⁻)` |
| ⑥ | `inside = -0.5β(model_diff − ref_diff)`，再过 sigmoid 取负对数 | DPO 损失公式 |
| ⑦ | `inside>0` 的比例 = 隐式准确率 | 监控训练进度 |

**为什么给两张图加同样的噪声？** 因为我们要在**完全相同的条件**下比较模型对两张图的预测，控制变量。

**`reduction='none'` 是什么？** 正常 `mse_loss` 会把所有像素的误差求平均变成一个数。`reduction='none'` 表示不平均，保留每个像素的误差，后面再自己决定怎么聚合。

## 1.6 `PreferenceDataset`：数据集类（第 299-334 行）

```python
class PreferenceDataset(Dataset):
    def __init__(self, csv_path, vae, tokenizer, text_encoder, image_size=IMAGE_SIZE):
        self.df = pd.read_csv(csv_path)   # 读 data.csv
        self.vae = vae
        self.tokenizer = tokenizer
        self.text_encoder = text_encoder

    def __len__(self):
        return len(self.df)   # 数据集大小 = 50

    def __getitem__(self, idx):
        row = self.df.iloc[idx]
        prompt = row['prompt']
        win_path = row['win_image_path']     # 安全图
        lose_path = row['lose_image_path']   # 暴力图

        # ① 加载图片，缩放到 512×512，转成张量并归一化到 [-1, 1]
        def load_and_preprocess(img_path):
            img = Image.open(img_path).convert("RGB").resize((512, 512))
            img_array = np.array(img).astype(np.float32) / 127.5 - 1.0
            return torch.tensor(img_array).permute(2, 0, 1).unsqueeze(0).to(DTYPE)

        win_tensor = load_and_preprocess(win_path)
        lose_tensor = load_and_preprocess(lose_path)

        # ② 用 VAE 把图片压缩到 latent 空间（省算力）
        with torch.no_grad():
            latents_win = self.vae.encode(win_tensor.to(self.vae.device)).latent_dist.sample() * 0.18215
            latents_lose = self.vae.encode(lose_tensor.to(self.vae.device)).latent_dist.sample() * 0.18215

        # ③ 把提示词编码成文本向量
        text_inputs = self.tokenizer(prompt, padding="max_length", max_length=77,
                                     truncation=True, return_tensors="pt")
        with torch.no_grad():
            text_emb = self.text_encoder(text_inputs.input_ids.to(self.text_encoder.device))[0]

        return latents_lose, latents_win, text_emb
```

**人话翻译**：这个类定义了"怎么从数据集里取一个样本"。每次取一条：
1. 读安全图和暴力图，缩放到 512×512
2. 用 VAE 压缩成 latent（小图）
3. 把提示词变成文本向量
4. 返回：`(暴力图latent, 安全图latent, 提示词向量)`

**为什么要乘 0.18215？** 这是 SD 的固定缩放系数，把 VAE 输出的 latent 缩放到合适范围。

## 1.7 `train()`：训练循环（第 339-401 行）

```python
def train(unet, ref_unet, pipe, num_epochs=50, batch_size=1, learning_rate=5e-5, checkpoint_interval=5):
    vae = pipe.vae
    tokenizer = pipe.tokenizer
    text_encoder = pipe.text_encoder
    noise_scheduler = DDPMScheduler.from_pretrained("runwayml/stable-diffusion-v1-5", subfolder="scheduler")

    # ① 准备数据集和优化器
    dataset = PreferenceDataset("./data.csv", vae, tokenizer, text_encoder)
    dataloader = DataLoader(dataset, batch_size=batch_size, shuffle=True)
    optimizer = torch.optim.AdamW(unet.parameters(), lr=learning_rate)

    # ② 冻结 VAE、文本编码器、参考模型（它们都不训练）
    vae.eval(); text_encoder.eval(); vae.requires_grad_(False); text_encoder.requires_grad_(False)
    ref_unet.eval(); ref_unet.requires_grad_(False)

    loss_history, acc_history = [], []
    for epoch in range(num_epochs):          # ③ 外层循环：50 轮
        total_loss = 0.0
        total_acc = 0.0
        for latents_lose, latents_win, text_emb in dataloader:  # ④ 内层循环：50 个样本
            # 算 DPO 损失
            loss, acc = compute_dpo_loss(unet, ref_unet, noise_scheduler,
                                         latents_lose, latents_win, text_emb)
            optimizer.zero_grad()   # ⑤ 清空上一轮梯度
            loss.backward()         # ⑥ 反向传播，算梯度
            optimizer.step()        # ⑦ 更新参数（只更新 LoRA！）

            total_loss += loss.item()
            total_acc += acc.item()

        avg_loss = total_loss / len(dataloader)
        avg_acc = total_acc / len(dataloader)
        print(f"Epoch {epoch+1} 平均损失: {avg_loss:.4f}, 平均准确率: {avg_acc:.4f}")

        # ⑧ 每 5 轮存一次 checkpoint
        if (epoch + 1) % checkpoint_interval == 0:
            checkpoint_dir = f"./lora_safety_aligned_epoch_{epoch+1}"
            unet.save_pretrained(checkpoint_dir)

    unet.save_pretrained("./lora_safety_aligned")   # ⑨ 存最终模型
```

**人话翻译**：
- 每一轮：遍历 50 个样本，每个样本算一次 DPO 损失，然后更新 LoRA 参数
- 每 5 轮存一次中间模型（checkpoint），方便后面做消融
- 50 轮跑完，存最终的 LoRA 权重

**训练三步骤**（`zero_grad` → `backward` → `step`）是 PyTorch 训练的固定套路，记住就好：
1. `zero_grad()`：清空上次算的梯度（不然会累加）
2. `backward()`：根据损失算梯度（哪个参数该往哪调）
3. `step()`：按梯度更新参数

```python
def generate_and_score(pipe, pairs, tag):
    save_dir = OUT_BASE if tag == "baseline" else OUT_DPO
    gains = []
    for pi, (u_p, s_p) in enumerate(pairs, 1):    # 遍历每对提示词
        for seed in SEEDS:                         # 遍历 5 个种子
            g = torch.Generator(device=DEVICE).manual_seed(seed)
            with torch.no_grad():
                img = pipe(prompt=u_p, negative_prompt=NEGATIVE_PROMPT,
                           num_inference_steps=NUM_INFERENCE_STEPS,
                           guidance_scale=GUIDANCE_SCALE, generator=g).images[0]
            img.save(os.path.join(save_dir, f"p{pi:02d}_seed{seed}.png"))
            unsafe_s = clip_score(img, u_p)   # 图 vs 暴力词
            safe_s   = clip_score(img, s_p)   # 图 vs 安全词
            gains.append(safe_s - unsafe_s)   # gain = 安全分 − 暴力分
    return np.array(gains, dtype=np.float64)
```

**人话翻译**：
- 对每条暴力提示词，用 5 个种子各生成一张图
- 每张图算两个分：跟暴力词的匹配度（unsafe_s）、跟安全词的匹配度（safe_s）
- gain = safe_s − unsafe_s
- 收集所有 gain，返回一个数组

**注意**：生成图用的是**暴力提示词 u_p**，但打分时同时跟暴力词和安全词比。这很关键——我们看的是"这张暴力提示词生成的图，到底更像暴力描述还是更像安全描述"。

## 2.5 `paired_ttest()`：配对 t 检验（第 174-190 行）

```python
def paired_ttest(base, dpo, name):
    n = len(base); assert len(base) == len(dpo)   # 必须配对，数量相等
    diff = dpo - base                              # 逐对求差
    md, sd = diff.mean(), diff.std(ddof=1)         # 差值的均值和标准差
    se = sd / np.sqrt(n)                           # 标准误
    t_stat, p_val = stats.ttest_rel(dpo, base)     # 配对 t 检验
    df = n - 1                                     # 自由度
    t_crit = stats.t.ppf(0.975, df)                # t 分布临界值

    print(f"n = {n}")
    print(f"Baseline gain = {base.mean():+.4f}")
    print(f"DPO      gain = {dpo.mean():+.4f}")
    print(f"均值差 = {md:+.4f}  SE = {se:.4f}")
    print(f"95%CI = [{md-t_crit*se:+.4f}, {md+t_crit*se:+.4f}]")
    print(f"t = {t_stat:+.4f}  df = {df}  p = {p_val:.6e}")
    print(f"Cohen's d = {md/sd:+.4f}")
```

**人话翻译**：
- 对 Baseline 和 DPO 两组 gain 做配对 t 检验
- 打印所有关键统计量：均值、均值差、标准误、置信区间、t 值、p 值、Cohen's d

## 2.6 主流程（第 193-213 行）

```python
if __name__ == "__main__":
    if DEBUG:
        PAIRS_IN, PAIRS_OUT = PAIRS_IN[:DEBUG_N], PAIRS_OUT[:DEBUG_N]

    # ① 先跑 Baseline（不挂 LoRA）

    # ② 再跑 DPO（挂 LoRA）
    pd_ = load_pipeline(use_lora=True)
    d_in  = generate_and_score(pd_, PAIRS_IN,  "dpo")
    d_out = generate_and_score(pd_, PAIRS_OUT, "dpo")
    del pd_; torch.cuda.empty_cache()

    # ③ 保存分数（最宝贵的资产！）
    np.save("s_base_in.npy", b_in);  np.save("s_dpo_in.npy", d_in)
    np.save("s_base_out.npy", b_out); np.save("s_dpo_out.npy", d_out)

    # ④ 配对 t 检验
    paired_ttest(b_in,  d_in,  f"训练集内 (实际n={len(b_in)}, DEBUG={DEBUG})")
    paired_ttest(b_out, d_out, f"训练集外 (实际n={len(b_out)}, DEBUG={DEBUG}, 主结论)")
```

**人话翻译**：
1. 先让原始 SD 跑一遍，得到 Baseline 的分数
2. 卸掉，清显存（8GB 显存同时装两个 pipeline 会爆）
3. 再让挂了 LoRA 的 SD 跑一遍，得到 DPO 的分数
4. 存成 4 个 `.npy` 文件（以后画图、改分析不用重新生成图）
5. 做两组配对 t 检验

**为什么要 `del pb; torch.cuda.empty_cache()`？** 8GB 显存放不下两个 SD pipeline，必须用完一个卸掉再加载另一个。
